frappe.ui.form.on('JobEstimation', {
    refresh: function(frm) {
        calculate_estimated_end_date(frm);
        generate_item_summary(frm); // Add item summary generation

        frm.add_custom_button(__('Calculate End Date'), function() {
            calculate_estimated_end_date(frm);
        });

        frm.add_custom_button(__('Generate Item Summary'), function() {
            generate_item_summary(frm);
        });
    },

    estimated_start_date: function(frm) {
        calculate_estimated_end_date(frm);
        update_operation_dates(frm);
    },

    siteinspection: function(frm) {
        if (frm.doc.siteinspection) {
            frappe.call({
                method: 'frappe.client.get',
                args: {
                    doctype: 'SiteInspection',
                    name: frm.doc.siteinspection
                },
                callback: function(r) {
                    console.log("📦 SiteInspection Response:", r.message);

                    if (r.message && r.message.site_dimensions) {
                        frm.clear_table('site_dimensions');
                        frm.clear_table('boq');

                        let all_images = [];
                        let entitySet = new Set();

                        r.message.site_dimensions.forEach((row) => {
                            let new_row = frm.add_child('site_dimensions');
                            new_row.no = row.no;
                            new_row.floor = row.floor;
                            new_row.room = row.room;
                            new_row.entity = row.entity;
                            new_row.area_name = row.area_name;
                            new_row.dimensionsunits = row.dimensionsunits;
                            new_row.media = row.media;

                            if (row.media) {
                                new_row.media_preview = `<img src="${row.media}" style="width: 60px; height: 60px; object-fit: cover; border-radius: 4px;">`;
                                all_images.push({
                                    image: row.media,
                                    caption: `${row.floor || ''} - ${row.room || ''} - ${row.area_name || ''} (Media)`,
                                    type: 'media'
                                });
                            }

                            if (row.entity && !entitySet.has(row.entity)) {
                                let boq_row = frm.add_child('boq');
                                boq_row.entity = row.entity;
                                boq_row.description = `${row.floor || ''} - ${row.room || ''} - ${row.area_name || ''}`;
                                entitySet.add(row.entity);
                            }
                        });

                        if (r.message.site_photos) {
                            all_images.push({
                                image: r.message.site_photos,
                                caption: `${r.message.name || 'Site Inspection'} (Site Photos)`,
                                type: 'site_photos'
                            });
                        }

                        if (r.message.measurement_sketch) {
                            all_images.push({
                                image: r.message.measurement_sketch,
                                caption: `${r.message.name || 'Site Inspection'} (Measurement Sketch)`,
                                type: 'measurement_sketch'
                            });
                        }

                        frm.refresh_field('site_dimensions');
                        frm.refresh_field('boq');
                        show_all_images_separately(frm, all_images);

                        frm.fields_dict.site_dimensions.grid.cannot_add_rows = true;
                        frm.fields_dict.site_dimensions.grid.cannot_delete_rows = true;

                        frappe.show_alert({ message: 'Site Dimensions loaded!', indicator: 'green' });
                    } else {
                        frappe.show_alert({ message: 'No Site Dimensions found!', indicator: 'yellow' });
                    }
                }
            });
        } else {
            frm.clear_table('site_dimensions');
            frm.clear_table('boq');
            frm.refresh_field('site_dimensions');
            frm.refresh_field('boq');
            if ($('#all-images-container').length) {
                $('#all-images-container').html('');
            }
        }
    }
});

frappe.ui.form.on('Materials Required_BOQ', {
    bom: function(frm, cdt, cdn) {
        let row = locals[cdt][cdn];
        if (row.bom) {
            frappe.call({
                method: 'frappe.client.get',
                args: {
                    doctype: 'BOM',
                    name: row.bom
                },
                callback: function(r) {
                    if (r.message) {
                        let bom_cost = r.message.total_cost || 0;

            generate_item_summary(frm); // Regenerate item summary
        }
    },

    complete_cost: function(frm) {
        update_total_estimation(frm);
        generate_item_summary(frm); // Regenerate item summary
    }
});

function update_total_estimation(frm) {
    let total = 0;
    (frm.doc.boq || []).forEach(row => {
        total += flt(row.complete_cost);
    });
    frm.set_value('total_estimation_amount', total);
    frm.refresh_field('total_estimation_amount');
}

function calculate_estimated_end_date(frm) {
    let total_time = 0;
    let start_date = frm.doc.estimated_start_date;

    if (!start_date) {
        frm.set_value('estimated_end_date', '');
        return;
    }

    if (frm.doc.tentative_schedule_for_execution && frm.doc.tentative_schedule_for_execution.length > 0) {
        frm.doc.tentative_schedule_for_execution.forEach(row => {
            if (row.time && !isNaN(parseFloat(row.time))) {
                total_time += parseFloat(row.time);
            }
        });
    }

    let total_days = Math.ceil(total_time / 8);

    if (total_days > 0) {
        let start_date_obj = new Date(start_date);
        if (!isNaN(start_date_obj.getTime())) {
            start_date_obj.setDate(start_date_obj.getDate() + total_days);
            let end_date = start_date_obj.toISOString().split('T')[0];
            frm.set_value('estimated_end_date', end_date);
        }
    } else {
        frm.set_value('estimated_end_date', '');
    }
}

function set_next_operation_date(frm) {
    let rows = frm.doc.tentative_schedule_for_execution || [];

    if (rows.length > 0) {
        let last_row = rows[rows.length - 1];
        let previous_row = rows[rows.length - 2];

        if (rows.length === 1) {
            if (frm.doc.estimated_start_date) {
                frappe.model.set_value(last_row.doctype, last_row.name, 'date', frm.doc.estimated_start_date);
            }
        } else if (previous_row && previous_row.date && previous_row.time) {
            let prev_start = new Date(previous_row.date);
            let prev_days = Math.ceil(parseFloat(previous_row.time || 0) / 8);
            prev_start.setDate(prev_start.getDate() + prev_days);
            let next_date = prev_start.toISOString().split('T')[0];
            frappe.model.set_value(last_row.doctype, last_row.name, 'date', next_date);
        }
    }
}

function update_operation_dates(frm) {
    let rows = frm.doc.tentative_schedule_for_execution || [];

    rows.forEach((row, index) => {
        if (index === 0) {
            if (frm.doc.estimated_start_date) {
                frappe.model.set_value(row.doctype, row.name, 'date', frm.doc.estimated_start_date);
            }
        } else {
            let previous_row = rows[index - 1];
            if (previous_row.date && previous_row.time) {
                let prev_start = new Date(previous_row.date);
                let prev_days = Math.ceil(parseFloat(previous_row.time || 0) / 8);
                prev_start.setDate(prev_start.getDate() + prev_days);
                let next_date = prev_start.toISOString().split('T')[0];
                frappe.model.set_value(row.doctype, row.name, 'date', next_date);
            }
        }
    });
}

// ✅ Show images in 3 sections
function show_all_images_separately(frm, images) {
    const categorized = {
        media: [],
        site_photos: [],
        measurement_sketch: []
    };

    images.forEach(img => {
        const fullUrl = img.image.startsWith('http') ? img.image : window.location.origin + img.image;
        categorized[img.type].push({
            url: fullUrl,
            caption: img.caption
        });
    });

    if ($('#all-images-container').length) {
        $('#all-images-container').remove();
    }

    let container = $('<div id="all-images-container" style="margin-top: 20px;"></div>');

    function renderCategory(title, items) {
        if (!items.length) return '';

        let html = `<h4 style="margin-bottom: 8px;">${title}</h4><div style="display:flex; flex-wrap:wrap; gap:15px;">`;
        items.forEach(img => {
            html += `
                <div style="border:1px solid #ddd; border-radius:8px; padding:10px; text-align:center; width:200px;">
                    <img src="${img.url}" style="width:180px; height:150px; object-fit:cover; border-radius:4px; cursor:pointer;" onclick="window.open('${img.url}', '_blank')">
                    <div style="margin-top: 8px; font-size: 12px; color: #666;">${img.caption}</div>
                </div>`;
        });
        html += '</div><br>';
        return html;
    }

    container.append(renderCategory('Media Preview', categorized.media));
    container.append(renderCategory('Site Images', categorized.site_photos));
    container.append(renderCategory('Measurement Sketch', categorized.measurement_sketch));

    $(frm.fields_dict.site_dimensions.grid.wrapper).after(container);
}

// 🆕 NEW: Generate Item and Labour Summary Table
function generate_item_summary(frm) {
    if (!frm.doc.boq || frm.doc.boq.length === 0) {
        remove_item_summary_table();
        return;
    }
    
    let bom_promises = [];
    let bom_data = {};
    let labour_data = {};
    
    // Collect all BOM data first
    frm.doc.boq.forEach(row => {
        if (row.bom) {
            let promise = frappe.call({
                method: 'frappe.client.get',
                args: {
                    doctype: 'BOM',
                    name: row.bom
                },
                callback: function(r) {
                    if (r.message) {
                        bom_data[row.bom] = {
                            entity: row.entity,
                            description: row.description,
                            bom_doc: r.message
                        };
                    }
                }
            });
            bom_promises.push(promise);
        }
    });
    
    // Collect labour data directly (no API calls needed)
    if (frm.doc.tentative_schedule_for_execution && frm.doc.tentative_schedule_for_execution.length > 0) {
        frm.doc.tentative_schedule_for_execution.forEach(row => {
            if (row.time) {
                labour_data[row.name || row.idx] = {
                    operation: row.operation,
                    date: row.date,
                    time: row.time,
                    employee: row.employee,
                    custom_hourly_rate: row.custom_hourly_rate,
                    remarks: row.remarks
                };
            }
        });
    }
    
    // Wait for all BOM data to be fetched, then process both BOM and labour data
    Promise.all(bom_promises).then(() => {
        process_bom_and_labour_items(frm, bom_data, labour_data);
    });
}

// Helper function to process both BOM and labour data
function process_bom_and_labour_items(frm, bom_data, labour_data) {
    // Process BOM items first
    process_bom_items(frm, bom_data);
    
    // Process labour data if available
    if (Object.keys(labour_data).length > 0) {
        process_labour_items(frm, labour_data);
    }
}


function process_bom_items(frm, bom_data) {
    let item_summary = {};
    let bom_breakdown = [];

    // Process each BOM
    Object.keys(bom_data).forEach(bom_name => {
        let bom_info = bom_data[bom_name];
        let bom_doc = bom_info.bom_doc;

        // Add BOM header info
        bom_breakdown.push({
            type: 'bom_header',
            bom_name: bom_name,
            entity: bom_info.entity,
            description: bom_info.description,
            total_cost: bom_doc.total_cost || 0
        });

        // Process BOM items
        if (bom_doc.items && bom_doc.items.length > 0) {
            bom_doc.items.forEach(item => {
                let item_code = item.item_code;
                let qty = item.qty || 0;
                let rate = item.rate || 0;
                let amount = item.amount || 0;

                // Add to BOM breakdown
                bom_breakdown.push({
                    type: 'bom_item',
                    bom_name: bom_name,
                    item_code: item_code,
                    item_name: item.item_name || '',
                    qty: qty,
                    uom: item.uom || '',
                    rate: rate,
                    amount: amount
                });

                // Add to item summary
                if (!item_summary[item_code]) {
                    item_summary[item_code] = {
                        item_name: item.item_name || '',
                        total_qty: 0,
                        uom: item.uom || '',
                        avg_rate: 0,
                        total_amount: 0,
                        bom_usage: []
                    };
                }

                item_summary[item_code].total_qty += qty;
                item_summary[item_code].total_amount += amount;
                item_summary[item_code].avg_rate = item_summary[item_code].total_amount / item_summary[item_code].total_qty;
                item_summary[item_code].bom_usage.push({
                    bom_name: bom_name,
                    entity: bom_info.entity,
                    qty: qty,
                    amount: amount
                });
            });
        }
    });

    display_item_summary_table(frm, item_summary, bom_breakdown);
}

// Helper function to process labour items
function process_labour_items(frm, labour_data) {
    let labour_summary = {};
    let labour_breakdown = [];

    // Process each labour entry
    Object.keys(labour_data).forEach(labour_id => {
        let labour_info = labour_data[labour_id];
        let hours = labour_info.time || 0;
        let rate = labour_info.custom_hourly_rate || 0;
        let amount = hours * rate;

        // Add to labour breakdown
        labour_breakdown.push({
            type: 'labour_item',
            labour_id: labour_id,
            operation: labour_info.operation || '',
            date: labour_info.date || '',
            employee: labour_info.employee || '',
            hours: hours,
            rate: rate,
            amount: amount,
            remarks: labour_info.remarks || ''
        });

        // Group by operation for summary
        let operation = labour_info.operation || 'Unknown Operation';
        if (!labour_summary[operation]) {
            labour_summary[operation] = {
                operation: operation,
                total_hours: 0,
                avg_rate: 0,
                total_amount: 0,
                employee_count: 0,
                employees: new Set(),
                entries: []
            };
        }

        labour_summary[operation].total_hours += hours;
        labour_summary[operation].total_amount += amount;
        labour_summary[operation].employees.add(labour_info.employee || 'Unknown');
        labour_summary[operation].employee_count = labour_summary[operation].employees.size;
        labour_summary[operation].avg_rate = labour_summary[operation].total_amount / labour_summary[operation].total_hours || 0;
        labour_summary[operation].entries.push({
            labour_id: labour_id,
            date: labour_info.date,
            employee: labour_info.employee,
            hours: hours,
            rate: rate,
            amount: amount
        });
    });

    // Convert employees Set to Array for easier handling
    Object.keys(labour_summary).forEach(operation => {
        labour_summary[operation].employees = Array.from(labour_summary[operation].employees);
    });

    display_labour_summary_table(frm, labour_summary, labour_breakdown);
}

        let labour = labour_summary[operation];
        total_labour_hours += labour.total_hours;
        total_labour_amount += labour.total_amount;
        
        labour_table_html += `
            <tr style="border-bottom: 1px solid #eee;">
                <td style="border: 1px solid #ddd; padding: 8px;">${row_index}</td>
                <td style="border: 1px solid #ddd; padding: 8px; font-weight: bold;">${labour.operation}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: right;">${format_currency(labour.total_hours, null, 2)} hrs</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: right;">${format_currency(labour.avg_rate)}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: right; font-weight: bold;">${format_currency(labour.total_amount)}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: center;">${labour.employee_count}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: center;">
                    <button onclick="show_labour_details('${operation}')" style="background: #007bff; color: white; border: none; padding: 4px 8px; border-radius: 3px; cursor: pointer; font-size: 10px;">Details</button>
                </td>
            </tr>`;
        row_index++;
    });

    labour_table_html += `
                    </tbody>
                    <tfoot>
                        <tr style="background-color: #f8f9fa; font-weight: bold;">
                            <td colspan="2" style="border: 1px solid #ddd; padding: 8px; text-align: left;">Total Labour:</td>
                            <td style="border: 1px solid #ddd; padding: 8px; text-align: right; color: #007bff;">${format_currency(total_labour_hours, null, 2)} hrs</td>
                            <td style="border: 1px solid #ddd; padding: 8px;"></td>
                            <td style="border: 1px solid #ddd; padding: 8px; text-align: right; color: #28a745;">${format_currency(total_labour_amount)}</td>
                            <td colspan="2" style="border: 1px solid #ddd; padding: 8px;"></td>
                        </tr>
                    </tfoot>
                </table>
            </div>
        </div>`;

    // Create Labour Breakdown Table
    let breakdown_table_html = `
        <div>
            <h4 style="margin-bottom: 15px; color: #333; font-weight: bold;">Labour Breakdown</h4>
            <div style="overflow-x: auto;">
                <table style="width: 100%; border-collapse: collapse; border: 1px solid #ddd; font-size: 12px;">
                    <thead>
                        <tr style="background-color: #f8f9fa;">
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: left;">Operation</th>
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: left;">Date</th>
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: left;">Employee</th>
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: right;">Hours</th>
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: right;">Rate/Hr</th>
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: right;">Amount</th>
                            <th style="border: 1px solid #ddd; padding: 8px; text-align: left;">Remarks</th>
                        </tr>
                    </thead>
                    <tbody>`;

    labour_breakdown.forEach(item => {
        breakdown_table_html += `
            <tr style="background-color: #f8f9fa;">
                <td style="border: 1px solid #ddd; padding: 8px; color: #495057;">${item.operation}</td>
                <td style="border: 1px solid #ddd; padding: 8px; color: #6c757d;">${item.date}</td>
                <td style="border: 1px solid #ddd; padding: 8px; color: #6c757d;">${item.employee}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: right;">${format_currency(item.hours, null, 2)}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: right;">${format_currency(item.rate)}</td>
                <td style="border: 1px solid #ddd; padding: 8px; text-align: right;">${format_currency(item.amount)}</td>
                <td style="border: 1px solid #ddd; padding: 8px; color: #6c757d;">${item.remarks}</td>
            </tr>`;
    });

    breakdown_table_html += `
                    </tbody>
                </table>
            </div>
        </div>`;

    container.html(labour_table_html + breakdown_table_html);

    // Store labour summary data globally for details popup
    window.labour_summary_data = labour_summary;

    // Add the container after the item summary table or material_total table
    if ($('#item-summary-container').length) {
        $('#item-summary-container').after(container);
    } else if (frm.fields_dict.material_total && frm.fields_dict.material_total.grid) {
        $(frm.fields_dict.material_total.grid.wrapper).after(container);
    } else {
        $(frm.wrapper).find('.form-layout').append(container);
    }
}

function remove_labour_summary_table() {
    if ($('#labour-summary-container').length) {
        $('#labour-summary-container').remove();
    }
}

// Global function to show labour details
window.show_labour_details = function(operation) {
    
    if (!window.labour_summary_data || !window.labour_summary_data[operation]) {
        frappe.msgprint('No details available for this operation.');
        return;
    }

    let labour = window.labour_summary_data[operation];
    let details_html = `
        <div style="margin-bottom: 15px;">
            <h5 style="margin-bottom: 10px; color: #333;">Operation: ${operation}</h5>
            <p><strong>Total Hours:</strong> ${format_currency(labour.total_hours, null, 2)} hrs</p>
            <p><strong>Average Rate:</strong> ${format_currency(labour.avg_rate)}/hr</p>
            <p><strong>Total Amount:</strong> ${format_currency(labour.total_amount)}</p>
            <p><strong>Employees:</strong> ${labour.employees.join(', ')}</p>
        </div>
        <div>
            <h6 style="margin-bottom: 10px; color: #555;">Labour Entries:</h6>
            <table style="width: 100%; border-collapse: collapse; border: 1px solid #ddd; font-size: 12px;">
                <thead>
                    <tr style="background-color: #f8f9fa;">
                        <th style="border: 1px solid #ddd; padding: 6px; text-align: left;">Date</th>
                        <th style="border: 1px solid #ddd; padding: 6px; text-align: left;">Employee</th>
                        <th style="border: 1px solid #ddd; padding: 6px; text-align: right;">Hours</th>
                        <th style="border: 1px solid #ddd; padding: 6px; text-align: right;">Rate/Hr</th>
                        <th style="border: 1px solid #ddd; padding: 6px; text-align: right;">Amount</th>
                    </tr>
                </thead>
                <tbody>`;

    labour.entries.forEach(entry => {
        details_html += `
            <tr>
                <td style="border: 1px solid #ddd; padding: 6px;">${entry.date}</td>
                <td style="border: 1px solid #ddd; padding: 6px;">${entry.employee}</td>
                <td style="border: 1px solid #ddd; padding: 6px; text-align: right;">${format_currency(entry.hours, null, 2)}</td>
                <td style="border: 1px solid #ddd; padding: 6px; text-align: right;">${format_currency(entry.rate)}</td>
                <td style="border: 1px solid #ddd; padding: 6px; text-align: right;">${format_currency(entry.amount)}</td>
            </tr>`;
    });

    details_html += `
                </tbody>
            </table>
        </div>`;

    frappe.msgprint({
        title: `Labour Details: ${operation}`,
        message: details_html,
        wide: true
    });
};
